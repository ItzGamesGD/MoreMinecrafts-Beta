    static final class T implements St01Eligibility.Terrain {
        final ChunkGenerator g; final RandomState rs; final ServerLevel level; final Map<Long, Integer> memo = new HashMap<>();
        long computes, ns;
        T(ServerLevel l) { level = l; g = l.getChunkSource().getGenerator(); rs = l.getChunkSource().randomState(); }
        public int surfaceHeight(int x, int z) {
            long k = ((long) x << 32) ^ (z & 0xFFFFFFFFL); Integer h = memo.get(k); if (h != null) return h;
            long t0 = System.nanoTime(); int v = g.getBaseHeight(x, z, Heightmap.Types.WORLD_SURFACE_WG, level, rs); ns += System.nanoTime() - t0; computes++;
            memo.put(k, v); return v;
        }
        public boolean isDesertCell(int qx, int qy, int qz) { return level.getUncachedNoiseBiome(qx, qy, qz).is(Biomes.DESERT); }
    }
    static RandomSpreadStructurePlacement placement(ServerLevel level) {
        return (RandomSpreadStructurePlacement) level.registryAccess().lookupOrThrow(Registries.STRUCTURE_SET).get(ST01).orElseThrow().value().placement();
    }
    record Policy(String name, int diff, int cut, int skirt) {}
    static final Policy D = new Policy("D_16_10_8", 16, 10, 8), B = new Policy("B_40_24_32", 40, 24, 32);

    /** Terrain verdict for a plane yf (same formulas as St01Eligibility.evaluate). */
    static String verdict(Policy p, int min, int max, int yf, T t, int cx, int cz) {
        int diff = max - min, cut = Math.max(0, max - yf), skirt = Math.max(0, yf - min);
        if (diff > p.diff) return "REJECT_HEIGHT_DIFFERENCE";
        if (cut > p.cut) return "REJECT_CUT";
        if (skirt > p.skirt) return "REJECT_SKIRT";
        if (!t.isDesertCell(Math.floorDiv(cx, 4), Math.floorDiv(yf, 4), Math.floorDiv(cz, 4))) return "REJECT_BIOME_STUB";
        return "ELIGIBLE";
    }

    static Map<String, Integer> planes(int[] sorted, int min, int max, Policy p) {
        int n = sorted.length; Map<String, Integer> m = new LinkedHashMap<>();
        m.put("median", sorted[n / 2]);
        long s = 0; for (int h : sorted) s += h; m.put("mean", (int) Math.round(s / (double) n));
        int k = n / 10; long ts = 0; for (int i = k; i < n - k; i++) ts += sorted[i]; m.put("trimmed10", (int) Math.round(ts / (double) (n - 2 * k)));
        m.put("p25", sorted[n / 4]); m.put("p40", sorted[n * 2 / 5]); m.put("p60", sorted[n * 3 / 5]); m.put("p75", sorted[n * 3 / 4]);
        m.put("midrange", Math.floorDiv(min + max, 2));
        // feasible-window clamp of the median: yf in [max - cut, min + skirt]
        int lo = max - p.cut, hi = min + p.skirt, med = sorted[n / 2];
        m.put("window_clamped_median", lo <= hi ? Math.max(lo, Math.min(hi, med)) : med);
        return m;
    }

    private static void coverage(ServerLevel level, Map<String, Object> res) {
        var pl = placement(level); long seed = level.getSeed(); int spacing = pl.spacing(), rc = 10240 / 16;
        int r0 = Math.floorDiv(-rc, spacing), r1 = Math.floorDiv(rc - 1, spacing);
        int potential = 0, desert = 0; List<Map<String, Object>> cands = new ArrayList<>();
        long t0 = System.nanoTime(), desertNs = 0, terrainNs = 0, hc = 0;
        for (int rx = r0; rx <= r1; rx++) for (int rz = r0; rz <= r1; rz++) {
            ChunkPos pc = pl.getPotentialStructureChunk(seed, rx * spacing, rz * spacing);
            if (Math.abs(pc.getMiddleBlockX()) > 10240 || Math.abs(pc.getMiddleBlockZ()) > 10240) continue;
            potential++;
            int cx = pc.getMinBlockX() + 8, cz = pc.getMinBlockZ() + 8;
            T t = new T(level);
            long a = System.nanoTime();
            boolean fd = St01Eligibility.fullFootprintDesert(t, cx, cz);
            desertNs += System.nanoTime() - a;
            if (!fd) { hc += t.computes; continue; }
            desert++;
            long b = System.nanoTime();
            int[] grid = new int[N * N]; int i = 0;
            for (int dx = -HALF; dx <= HALF; dx++) for (int dz = -HALF; dz <= HALF; dz++) grid[i++] = t.surfaceHeight(cx + dx, cz + dz);
            terrainNs += System.nanoTime() - b;
            int[] sorted = grid.clone(); Arrays.sort(sorted);
            int min = sorted[0], max = sorted[sorted.length - 1];
            // outside ring (one block outside the footprint)
            int[] ring = new int[4 * (N + 2) - 4]; int ri = 0;
            for (int d = -HALF - 1; d <= HALF + 1; d++) { ring[ri++] = t.surfaceHeight(cx + d, cz - HALF - 1); ring[ri++] = t.surfaceHeight(cx + d, cz + HALF + 1); }
            for (int d = -HALF; d <= HALF; d++) { ring[ri++] = t.surfaceHeight(cx - HALF - 1, cz + d); ring[ri++] = t.surfaceHeight(cx + HALF + 1, cz + d); }
            hc += t.computes;
            Map<String, Object> c = new LinkedHashMap<>();
            c.put("chunk", pc.x() + "," + pc.z()); c.put("cx", cx); c.put("cz", cz); c.put("min", min); c.put("max", max); c.put("diff", max - min);
            c.put("q", new int[]{sorted[0], sorted[sorted.length / 10], sorted[sorted.length / 4], sorted[sorted.length / 2], sorted[sorted.length * 3 / 4], sorted[sorted.length * 9 / 10], sorted[sorted.length - 1]});
            Map<String, Object> pol = new LinkedHashMap<>();
            for (Policy p : List.of(D, B)) {
                Map<String, Object> per = new LinkedHashMap<>();
                for (var e : planes(sorted, min, max, p).entrySet()) {
                    int yf = e.getValue();
                    long exc = 0, expo = 0, cliff = 0, exposeMax = 0, cliffMax = 0; int excCols = 0;
                    for (int h : grid) { if (h - 1 > yf) { exc += h - 1 - yf; excCols++; } }
                    for (int h : ring) { if (h < yf) { expo += yf - h; exposeMax = Math.max(exposeMax, yf - h); } if (h - 1 > yf) { cliff += h - 1 - yf; cliffMax = Math.max(cliffMax, h - 1 - yf); } }
                    Map<String, Object> r = new LinkedHashMap<>();
                    r.put("yf", yf); r.put("cut", Math.max(0, max - yf)); r.put("skirt", Math.max(0, yf - min));
                    r.put("verdict", verdict(p, min, max, yf, t, cx, cz));
                    r.put("excavation_blocks", exc); r.put("excavation_columns", excCols);
                    r.put("exposed_body_face_sum", expo); r.put("exposed_body_face_max", exposeMax);
                    r.put("boundary_cliff_face_sum", cliff); r.put("boundary_cliff_face_max", cliffMax);
                    per.put(e.getKey(), r);
                }
                pol.put(p.name, per);
            }
            c.put("policies", pol);
            String vb = (String) ((Map<?, ?>) ((Map<?, ?>) pol.get(B.name)).get("median")).get("verdict");
            String vbw = (String) ((Map<?, ?>) ((Map<?, ?>) pol.get(B.name)).get("window_clamped_median")).get("verdict");
            if (vb.equals("ELIGIBLE") || vbw.equals("ELIGIBLE")) {
                int yf = (Integer) ((Map<?, ?>) ((Map<?, ?>) pol.get(B.name)).get("median")).get("yf");
                c.put("overlap", overlap(level, cx, cz, yf, grid));
                ByteBuffer bb = ByteBuffer.allocate(grid.length * 2).order(ByteOrder.LITTLE_ENDIAN); for (int h : grid) bb.putShort((short) h);
                c.put("grid_i16le_b64", Base64.getEncoder().encodeToString(bb.array()));
            }
            cands.add(c);
        }
        res.put("potential", potential); res.put("full_desert", desert);
        res.put("desert_stage_ms", desertNs / 1e6); res.put("terrain_stage_ms", terrainNs / 1e6); res.put("height_computes", hc); res.put("total_ms", (System.nanoTime() - t0) / 1e6);
        res.put("candidates", cands);
    }

    /** Vanilla structure starts whose pieces intersect the ST-01 volume, the excavation cells, or the one-block outside ring. */
    static List<Map<String, Object>> overlap(ServerLevel level, int cx, int cz, int yf, int[] grid) {
        var reg = level.registryAccess().lookupOrThrow(Registries.STRUCTURE);
        BoundingBox vol = new BoundingBox(cx - HALF, yf - 32, cz - HALF, cx + HALF, yf + 39, cz + HALF);
        Map<String, StructureStart> found = new LinkedHashMap<>();
        for (int x = (cx - HALF) >> 4; x <= (cx + HALF) >> 4; x++) for (int z = (cz - HALF) >> 4; z <= (cz + HALF) >> 4; z++) {
            ChunkAccess ca = level.getChunk(x, z, ChunkStatus.STRUCTURE_REFERENCES);
            for (var e : ca.getAllReferences().entrySet()) for (long ref : e.getValue()) {
                ChunkPos sc = ChunkPos.unpack(ref);
                StructureStart s = level.getChunk(sc.x(), sc.z(), ChunkStatus.STRUCTURE_STARTS).getStartForStructure(e.getKey());
                if (s != null && s.isValid() && s.getBoundingBox().intersects(vol.inflatedBy(1))) found.put(reg.getKey(e.getKey()) + "@" + sc.x() + "," + sc.z(), s);
            }
        }
        List<Map<String, Object>> out = new ArrayList<>();
        for (var e : found.entrySet()) {
            long volCells = 0, excCells = 0, ringCells = 0; int pieces = 0;
            for (var piece : e.getValue().getPieces()) {
                BoundingBox pb = piece.getBoundingBox();
                if (!pb.intersects(vol.inflatedBy(1))) continue;
                pieces++;
                int x0 = Math.max(pb.minX(), vol.minX()), x1 = Math.min(pb.maxX(), vol.maxX()), z0 = Math.max(pb.minZ(), vol.minZ()), z1 = Math.min(pb.maxZ(), vol.maxZ());
                int y0 = Math.max(pb.minY(), vol.minY()), y1 = Math.min(pb.maxY(), vol.maxY());
                if (x0 <= x1 && z0 <= z1 && y0 <= y1) volCells += (long) (x1 - x0 + 1) * (z1 - z0 + 1) * (y1 - y0 + 1);
                for (int x = x0; x <= x1; x++) for (int z = z0; z <= z1; z++) {
                    int h = grid[(x - cx + HALF) * N + (z - cz + HALF)];
                    int ey0 = Math.max(yf + 1, pb.minY()), ey1 = Math.min(h - 1, pb.maxY());
                    if (ey1 >= ey0) excCells += ey1 - ey0 + 1;
                }
                // pieces reaching the one-block ring outside the footprint (where a later edge blend would write)
                BoundingBox ringBox = new BoundingBox(cx - HALF - 1, vol.minY(), cz - HALF - 1, cx + HALF + 1, vol.maxY(), cz + HALF + 1);
                if (pb.intersects(ringBox) && (pb.minX() < cx - HALF || pb.maxX() > cx + HALF || pb.minZ() < cz - HALF || pb.maxZ() > cz + HALF)) ringCells++;
            }
            Map<String, Object> r = new LinkedHashMap<>();
            r.put("structure", e.getKey()); r.put("step", e.getValue().getStructure().step().name()); r.put("pieces_touching", pieces);
            r.put("piece_box_cells_inside_st01_volume", volCells); r.put("piece_box_cells_in_excavation", excCells); r.put("pieces_crossing_footprint_edge", ringCells);
            out.add(r);
        }
        return out;
    }

    // ------------------------------------------------------------- search latency to the first eligible anchor (eligibility only)
    private static void search(ServerLevel level, Map<String, Object> res) {
        int radius = Integer.getInteger("mm.st01.c3a.radius", 20);
        var pl = placement(level); long seed = level.getSeed(); int spacing = pl.spacing();
        List<Map<String, Object>> runs = new ArrayList<>();
        String[] order = System.getProperty("mm.st01.c3a.order", "A,B,C,C,B,A").split(",");
        for (String v : order) {
            long t0 = System.nanoTime(); int cand = 0; String found = null; int ring = -1; long hc = 0;
            outer:
            for (int r = 0; r <= radius; r++) for (int k = -r; k <= r; k++) { boolean ek = k == -r || k == r;
                for (int l = -r; l <= r; l++) { if (!ek && l != -r && l != r) continue;
                    ChunkPos pc = pl.getPotentialStructureChunk(seed, spacing * k, spacing * l); cand++;
                    T t = new T(level); int cx = pc.getMinBlockX() + 8, cz = pc.getMinBlockZ() + 8;
                    boolean ok = switch (v) { case "A" -> evalProd(t, cx, cz, D); case "B" -> evalProd(t, cx, cz, B); default -> evalCheap(t, cx, cz, B); };
                    hc += t.computes;
                    if (ok) { found = pc.x() + "," + pc.z(); ring = r; break outer; }
                } }
            Map<String, Object> m = new LinkedHashMap<>();
            m.put("variant", v); m.put("ms", (System.nanoTime() - t0) / 1e6); m.put("candidates", cand); m.put("height_computes", hc); m.put("first_eligible", found); m.put("ring", ring);
            runs.add(m);
        }
        res.put("radius", radius); res.put("runs", runs);
    }
    /** Production order: Desert sweep, all 6561 heights, median plane, limits, stub. */
    static boolean evalProd(T t, int cx, int cz, Policy p) {
        if (!St01Eligibility.fullFootprintDesert(t, cx, cz)) return false;
        int[] h = new int[N * N]; int i = 0; for (int dx = -HALF; dx <= HALF; dx++) for (int dz = -HALF; dz <= HALF; dz++) h[i++] = t.surfaceHeight(cx + dx, cz + dz);
        Arrays.sort(h); return verdict(p, h[0], h[h.length - 1], h[h.length / 2], t, cx, cz).equals("ELIGIBLE");
    }
    /** P8-D2-R1 exact-equivalent cheap-first order (lattice + range early exit), same verdict. */
    static boolean evalCheap(T t, int cx, int cz, Policy p) {
        if (!St01Eligibility.fullFootprintDesert(t, cx, cz)) return false;
        int mn = Integer.MAX_VALUE, mx = Integer.MIN_VALUE;
        for (int dx = -HALF; dx <= HALF; dx += 8) for (int dz = -HALF; dz <= HALF; dz += 8) { int h = t.surfaceHeight(cx + dx, cz + dz); mn = Math.min(mn, h); mx = Math.max(mx, h); if (mx - mn > p.diff) return false; }
        int[] h = new int[N * N]; int i = 0;
        for (int dx = -HALF; dx <= HALF; dx++) for (int dz = -HALF; dz <= HALF; dz++) { int v = t.surfaceHeight(cx + dx, cz + dz); h[i++] = v; mn = Math.min(mn, v); mx = Math.max(mx, v); if (mx - mn > p.diff) return false; }
        Arrays.sort(h); return verdict(p, h[0], h[h.length - 1], h[h.length / 2], t, cx, cz).equals("ELIGIBLE");
    }

            m.put("footprint_open_cells_at_or_below_plane", footVoid); m.put("footprint_fluid_cells", footFluid); m.put("footprint_water", footWater); m.put("footprint_lava", footLava);
            m.put("open_cells_directly_below_floor", floorVoid);
            m.put("ring_air_cells_at_or_below_plane", ringVoidBelow); m.put("ring_fluid_cells_at_or_below_plane", ringFluidBelow);
            m.put("ring_fluid_cells_above_plane", ringFluidAbove); m.put("ring_water_above_plane", ringWaterAbove); m.put("ring_lava_above_plane", ringLavaAbove);
            m.put("st01_start_already_present", st01Present); m.put("vanilla_structures", new ArrayList<>(vnames)); m.put("vanilla_piece_boxes", vboxes.size());
            m.put("foreign_blocks_in_piece_boxes_at_or_below_plane", foreignBelow); m.put("foreign_blocks_in_piece_boxes_in_clearance_band", foreignClearBand);
            m.put("foreign_blocks_in_piece_boxes_above_clearance", foreignAboveClear);
            m.put("foreign_block_ids", foreignIds.entrySet().stream().sorted((p1, p2) -> p2.getValue() - p1.getValue()).limit(12).map(e -> e.getKey() + "=" + e.getValue()).toList());
            m.put("post_correction_leak_faces", leakFaces); m.put("post_correction_leak_water", leakWater); m.put("post_correction_leak_lava", leakLava);
            out.add(m);
        }
        res.put("corrections", out);
    }

    /** Not natural Desert terrain and not ST-01 sandstone palette: treated as a vanilla-structure block. */
    static boolean foreign(net.minecraft.world.level.block.state.BlockState st) {
        String id = net.minecraft.core.registries.BuiltInRegistries.BLOCK.getKey(st.getBlock()).getPath();
        if (id.contains("sandstone") || id.equals("sand") || id.equals("red_sand") || id.equals("stone") || id.equals("deepslate") || id.equals("dirt")
                || id.equals("gravel") || id.equals("clay") || id.equals("calcite") || id.equals("tuff") || id.equals("granite") || id.equals("diorite") || id.equals("andesite")
                || id.endsWith("_ore") || id.equals("dead_bush") || id.equals("cactus") || id.equals("short_dry_grass") || id.equals("tall_dry_grass") || id.equals("bedrock")
                || id.equals("glow_lichen") || id.contains("dripstone") || id.equals("moss_block")) return false;
        return true;
    }
